<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\SoftDeletes;

/**
 * @property-read \Illuminate\Database\Eloquent\Collection<int, \App\Models\AccountPayable> $accountPayables
 * @property-read int|null $account_payables_count
 * @property-read \Illuminate\Database\Eloquent\Collection<int, Supplier> $branches
 * @property-read int|null $branches_count
 * @property-read Supplier|null $parent
 * @property-read \Illuminate\Database\Eloquent\Collection<int, \App\Models\Transaction> $transactions
 * @property-read int|null $transactions_count
 * @property-read array $family_ids
 * @method static \Illuminate\Database\Eloquent\Builder<static>|Supplier newModelQuery()
 * @method static \Illuminate\Database\Eloquent\Builder<static>|Supplier newQuery()
 * @method static \Illuminate\Database\Eloquent\Builder<static>|Supplier onlyTrashed()
 * @method static \Illuminate\Database\Eloquent\Builder<static>|Supplier query()
 * @method static \Illuminate\Database\Eloquent\Builder<static>|Supplier withTrashed(bool $withTrashed = true)
 * @method static \Illuminate\Database\Eloquent\Builder<static>|Supplier withoutTrashed()
 * @mixin \Illuminate\Database\Eloquent\Builder
 */
class Supplier extends Model
{
    use HasFactory, SoftDeletes;

    protected $guarded = [];

    protected function casts(): array
    {
        return [
            'is_trusted'  => 'boolean',
            'is_personal' => 'boolean',
            'is_active'   => 'boolean',
            'order_days'  => 'array',
        ];
    }

    public function transactions()
    {
        return $this->hasMany(Transaction::class);
    }

    public function parent()
    {
        return $this->belongsTo(Supplier::class, 'parent_id');
    }

    public function branches()
    {
        return $this->hasMany(Supplier::class, 'parent_id');
    }

    public function accountPayables()
    {
        return $this->hasMany(\App\Models\AccountPayable::class, 'supplier_id');
    }

    /**
     * 🚀 ACESSOR MÁGICO: Retorna os IDs da Matriz e de todas as Filiais.
     */
    public function getFamilyIdsAttribute(): array
    {
        // 1. Descobre o ID do "Pai" (Se não tiver parent_id, ele próprio é a matriz)
        $matrixId = $this->parent_id ?? $this->id;

        // 2. Busca todos os "Filhos" (Filiais) que apontam para esta Matriz
        $branchIds = self::where('parent_id', $matrixId)->pluck('id')->toArray();

        // 3. Junta a Matriz com as Filiais num único Array e limpa duplicações
        return array_unique(array_merge([$matrixId], $branchIds));
    }

    // 🚀 A TRAVA ABSOLUTA DO BANCO DE DADOS
        protected static function booted()
        {
            static::saving(function ($supplier) {
                if (!empty($supplier->document)) {
                    $exists = self::where('document', $supplier->document)
                        ->where('id', '!=', $supplier->id)
                        ->exists();

                    if ($exists) {
                        throw \Illuminate\Validation\ValidationException::withMessages([
                            'document' => "O CNPJ/CPF '{$supplier->document}' já está em uso por outro fornecedor na base de dados."
                        ]);
                    }
                }
            });

            // Se for cadastrada uma nova filial de uma matriz confiável, herda automaticamente
            static::creating(function ($supplier) {
                if (!empty($supplier->parent_id) && !isset($supplier->is_trusted)) {
                    $parent = self::find($supplier->parent_id);
                    if ($parent && (bool) $parent->is_trusted) {
                        $supplier->is_trusted = true;
                    }
                }
            });

            // Ao salvar alterações em uma matriz, propaga is_trusted para todas as suas filiais
            static::saved(function ($supplier) {
                if ($supplier->wasChanged('is_trusted')) {
                    $supplier->branches()->update(['is_trusted' => (bool) $supplier->is_trusted]);
                }
            });
        }
}

<?php

namespace App\Filament\Resources\Suppliers\Tables;

use Filament\Tables\Table;
use Filament\Tables;
use App\Models\Supplier;
use App\Filament\Resources\Suppliers\Actions\MergeSuppliersBulkAction;
use Illuminate\Database\Eloquent\Builder;

class SuppliersTable
{
    public static function configure(Table $table): Table
    {
        return $table
            ->columns([
                Tables\Columns\TextColumn::make("name")
                    ->label("Fornecedor (Nome/Fantasia)")
                    ->searchable()
                    ->weight("bold")
                    ->sortable(),

                Tables\Columns\TextColumn::make("document_name")
                    ->label("Razão Social (Fiscal)")
                    ->searchable()
                    ->toggleable(isToggledHiddenByDefault: true),

                Tables\Columns\TextColumn::make("document")
                    ->label("CNPJ/CPF")
                    ->searchable(),

                Tables\Columns\TextColumn::make("type")
                    ->label("Tipo")
                    ->badge()
                    ->formatStateUsing(
                        fn(string $state): string => match ($state) {
                            "physical" => "Físico",
                            "virtual" => "Virtual",
                            default => $state,
                        },
                    )
                    ->colors([
                        "success" => "virtual",
                        "warning" => "physical",
                    ]),

                Tables\Columns\TextColumn::make("branches_count")
                    ->label("Filiais")
                    ->counts("branches")
                    ->badge()
                    ->color("gray")
                    ->sortable(),

                Tables\Columns\ToggleColumn::make("is_trusted")
                    ->label("Confiável")
                    ->tooltip("Manifesta Ciência da Emissão automaticamente na SEFAZ e dá entrada no estoque sem operador se todos os itens estiverem automatizados.")
                    ->afterStateUpdated(function (Supplier $record, bool $state) {
                        if ($record->branches()->exists()) {
                            $record->branches()->update(['is_trusted' => $state]);
                        }
                    }),

                Tables\Columns\ToggleColumn::make("is_active")
                    ->label("Ativo"),
            ])
            ->filters([
                Tables\Filters\Filter::make("only_matrix")
                    ->label("Apenas Matrizes")
                    ->default(true)
                    ->query(fn (Builder $query): Builder => $query->whereNull("parent_id")),

                Tables\Filters\SelectFilter::make("type")
                    ->label("Filtrar por Tipo")
                    ->options([
                        "physical" => "Físico",
                        "virtual" => "Virtual",
                    ]),
            ])
            ->actions([
                \Filament\Actions\EditAction::make()->iconButton(),
            ])
            ->bulkActions([
                \Filament\Actions\BulkActionGroup::make([
                    MergeSuppliersBulkAction::make(),
                    \Filament\Actions\DeleteBulkAction::make(),
                ]),
            ]);
    }
}

namespace App\Filament\Resources\Suppliers\RelationManagers;

use Filament\Resources\RelationManagers\RelationManager;
use Filament\Tables;
use Filament\Tables\Table;
use Illuminate\Database\Eloquent\Builder;

class FinancialRelationManager extends RelationManager
{
    protected static string $relationship = 'accountPayables'; // Usaremos a tabela de contas a pagar como base
    protected static ?string $title = 'Extrato Financeiro (NFes e Pagamentos)';

    public function table(Table $table): Table
    {
        return $table
            ->recordTitleAttribute('description')
            ->columns([
                Tables\Columns\TextColumn::make('due_date')->label('Vencimento')->date('d/m/Y')->sortable(),
                Tables\Columns\TextColumn::make('description')->label('Descrição')->searchable(),
                Tables\Columns\TextColumn::make('amount')->label('Valor')->money('BRL'),
                Tables\Columns\TextColumn::make('status')
                    ->label('Situação')
                    ->badge()
                    ->color(fn (string $state): string => match ($state) {
                        'pending' => 'warning',
                        'paid' => 'success',
                        'cancelled' => 'danger',
                    })
                    ->formatStateUsing(fn ($state) => match($state) { 
                        'pending' => 'Pendente', 
                        'paid' => 'Pago', 
                        'cancelled' => 'Cancelado' 
                    }),
            ])
            ->filters([
                Tables\Filters\SelectFilter::make('status')
                    ->options(['pending' => 'Pendente', 'paid' => 'Pago']),
            ])
            ->defaultSort('due_date', 'desc');
    }
}
