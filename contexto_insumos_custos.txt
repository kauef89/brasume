use App\Enums\ProductType;
use App\Observers\ProductObserver;
use Illuminate\Database\Eloquent\Attributes\ObservedBy;
use Illuminate\Database\Eloquent\Builder;

#[ObservedBy([ProductObserver::class])]
class Ingredient extends Product
{
    protected $table = 'products';

    /**
     * Força as chaves estrangeiras de relacionamentos hasMany (ex: costHistories, inventoryMovements)
     * a utilizarem product_id na tabela filha, e não ingredient_id.
     */
    public function getForeignKey(): string
    {
        return 'product_id';
    }

    public function getMorphClass(): string
    {
        return Product::class;
    }

    protected static function booted(): void
    {
        parent::booted();

        // 🚀 Escopo Global: este modelo acessa estritamente Insumos e Pré-preparos
        static::addGlobalScope('ingredient_type', function (Builder $builder) {
            $builder->whereIn('type', [
                ProductType::INGREDIENT->value,
                ProductType::MANUFACTURED->value,
            ]);
        });

        // 🚀 Valores padrão na criação
        static::creating(function (Ingredient $ingredient) {
            if (empty($ingredient->type)) {
                $ingredient->type = ProductType::INGREDIENT->value;
            }
            $ingredient->show_on_menu = false;
            $ingredient->show_on_customer_app = false;
        });
    }
}

                            "product_id",

                            ->first();

                            ? $latestHistory->unit_cost
                            : $record->base_cost;

                            number_format((float) $cost, 5, ",", ".");

                    ->color("success"),
            ])
            ->filters([
                Tables\Filters\SelectFilter::make("category_id")
                    ->label("Filtrar Categoria")
                    ->options(
                        fn() => Category::where("type", "internal")->pluck(
                            "name",
                            "id",
                        ),
                    ),
            ])
            ->actions([
                Actions\ViewAction::make()->iconButton(),
                Actions\EditAction::make()->iconButton(),
            ])

<?php

namespace App\Services\Purchases;

use App\Events\CatalogUpdated;
use App\Models\GoodsReceiptItem;

class InvoiceItemProcessor
{
    public function resolveStorageLocationForProcurement(Product $product): ?int
    {
        if (!$product->track_inventory) {
            return null;
        }

        $locations = DB::table('product_storage_locations as psl')
            ->join('storage_locations as sl', 'sl.id', '=', 'psl.storage_location_id')
            ->where('psl.product_id', $product->id)
            ->whereNull('sl.deleted_at')
            ->where('sl.is_active', 1)
            ->select('psl.*', 'sl.type as location_type')
            ->get();

        $storageLoc = $locations->firstWhere('location_type', 'storage');
        $displayLoc = $locations->firstWhere('location_type', 'display');

        // CENÁRIO 1: Possui Armazenamento e Exposição -> Vai 100% para o Armazenamento
        if ($storageLoc && $displayLoc) {
            return (int) $storageLoc->storage_location_id;
        }

        // CENÁRIO 2: Possui apenas local de armazenamento
        if ($storageLoc) {
            return (int) $storageLoc->storage_location_id;
        }

        // CENÁRIO 3: Possui apenas exposição com local de reposição de origem
        if ($displayLoc && !empty($displayLoc->source_location_id)) {
            $sourceId = (int) $displayLoc->source_location_id;
            DB::table('product_storage_locations')->updateOrInsert(
                ['product_id' => $product->id, 'storage_location_id' => $sourceId],
                ['updated_at' => now()]
            );
            return $sourceId;
        }

        // CENÁRIO 4: Revenda sem storage explícito -> Vincula ao Estoque de Armazenamento padrão
        $isResale = $product->type === 'resale' || (class_exists(\App\Enums\ProductType::class) && $product->type === \App\Enums\ProductType::RESALE->value);
        if ($isResale) {
            $defaultStorage = DB::table('storage_locations')
                ->where('type', 'storage')
                ->where('is_active', 1)
                ->whereNull('deleted_at')
                ->orderBy('sort_order')
                ->first();

            if ($defaultStorage) {
                DB::table('product_storage_locations')->updateOrInsert(
                    ['product_id' => $product->id, 'storage_location_id' => $defaultStorage->id],
                    ['updated_at' => now()]
                );

                if ($displayLoc && empty($displayLoc->source_location_id)) {
                    DB::table('product_storage_locations')
                        ->where('product_id', $product->id)
                        ->where('storage_location_id', $displayLoc->storage_location_id)
                        ->update(['source_location_id' => $defaultStorage->id, 'updated_at' => now()]);
                }

                return (int) $defaultStorage->id;
            }
        }

        return $locations->isNotEmpty() ? (int) $locations->first()->storage_location_id : null;
    }

    public function calculateSmartConversion(?int $productId, ?int $supplierId, ?string $currentXmlName, ?string $xmlUomRaw): array
    {
        $result = [
            'factor' => 1.0,
            'source' => 'manual',
            'notification' => null,
        ];

        if (!$productId || !$currentXmlName) {
            return $result;
        }

        if ($supplierId) {
            $lastPurchase = ProductCostHistory::where('supplier_id', $supplierId)
                ->where('product_id', $productId)
                ->where('xml_name', $currentXmlName)
                ->latest('id')
                ->first();

            if ($lastPurchase && $lastPurchase->conversion_factor) {
                $factor = (float) $lastPurchase->conversion_factor;
                $result['factor'] = $factor;
                $result['source'] = 'history';
                return $result;
            }
        }

        $product = Product::find($productId);
        $internalUom = strtoupper((string) ($product?->unit_of_measure ?? ''));
        $xmlUom = strtoupper(preg_replace('/[^A-Z]/', '', (string) ($xmlUomRaw ?? '')));

        $isIdentical = ($xmlUom === $internalUom)
            || (in_array($xmlUom, ['KG', 'KIL']) && $internalUom === 'KG')
            || (in_array($xmlUom, ['L', 'LT', 'LIT']) && $internalUom === 'LT')
            || (in_array($xmlUom, ['UN', 'UND']) && $internalUom === 'UN');

        if ($isIdentical) {
            $result['factor'] = 1.0;
            $result['source'] = 'identical';
            return $result;
        }

        $factor = 1.0;
        $autoMath = false;

        if (in_array($xmlUom, ['KG', 'KIL']) && $internalUom === 'G') {
            $factor = 1000.0;
            $autoMath = true;
        } elseif ($xmlUom === 'G' && in_array($internalUom, ['KG', 'KIL'])) {
            $factor = 0.001;
            $autoMath = true;
        } elseif (in_array($xmlUom, ['L', 'LT', 'LIT']) && $internalUom === 'ML') {
            $factor = 1000.0;
            $autoMath = true;
        } elseif ($xmlUom === 'ML' && in_array($internalUom, ['L', 'LT', 'LIT'])) {
            $factor = 0.001;
            $autoMath = true;
        }

        if ($autoMath) {
            $result['factor'] = $factor;
            $result['source'] = 'auto_math';
        }

        return $result;
    }

    public function processDirectItem(array $item, ?string $invoiceKey, ?int $supplierId, bool $autoApprove = false): void
    {
        $processQty = (float) ($item['xml_qty'] ?? 0);
        $invoiceCost = (float) ($item['xml_cost'] ?? 0);
        $invoiceUom = (string) ($item['xml_uom'] ?? '');
        $destination = $item['destination_type'] ?? (!empty($item['is_personal']) ? 'personal' : 'stock');
        $receiptItemId = $item['goods_receipt_item_id'] ?? null;

        // 🚀 ROTA 1: USO PESSOAL
        if ($destination === 'personal') {
            PersonalPurchase::create([
                'nfe_key'     => $invoiceKey,
                'supplier_id' => $supplierId,
                'xml_name'    => $item['xml_name'],
                'quantity'    => $processQty,
                'unit_cost'   => $invoiceCost,
                'total_cost'  => round($processQty * $invoiceCost, 4),
                'auto_approve'=> (bool) $autoApprove,
            ]);
            return;
        }

        // 🚀 ROTA 2: PATRIMÔNIO / ATIVO IMOBILIZADO
        if ($destination === 'asset') {
            CompanyAsset::create([
                'name'             => !empty($item['asset_name']) ? $item['asset_name'] : $item['xml_name'],
                'asset_tag'        => !empty($item['asset_tag']) ? $item['asset_tag'] : null,
                'location'         => !empty($item['asset_location']) ? $item['asset_location'] : null,
                'description'      => "Adquirido via NF-e {$invoiceKey} - Item XML: {$item['xml_name']}",
                'acquisition_cost' => round($processQty * $invoiceCost, 2),
                'acquisition_date' => now()->toDateString(),
                'warranty_until'   => !empty($item['warranty_until']) ? $item['warranty_until'] : null,
                'supplier_id'      => $supplierId,
                'nfe_key'          => $invoiceKey,
                'status'           => 'active',
            ]);
            return;
        }

        // 🚀 ROTA 3: ESTOQUE / INSUMO / REVENDA
        $productId = $item['product_id'];
        $conversionFactor = (float) ($item['conversion_factor'] ?? 1.0);
        if ($conversionFactor <= 0) $conversionFactor = 1.0;

        $actualQty = round($processQty * $conversionFactor, 4);
        $actualUnitCost = round($invoiceCost / $conversionFactor, 5);

        $product = Product::find($productId);
        $inventoryService = app(InventoryService::class);

        if ($product) {
            $product->updateQuietly(['base_cost' => $actualUnitCost]);

            if ($receiptItemId) {
                $receiptItem = GoodsReceiptItem::find($receiptItemId);
                if ($receiptItem) {
                    $receiptQty = (float) $receiptItem->quantity;
                    $delta = round($actualQty - $receiptQty, 4);

                    if ($delta > 0) {
                        $targetLocationId = $this->resolveStorageLocationForProcurement($product);
                        $inventoryService->addStock($product->fresh(), $delta, 'nfe_reconcile_surplus', null, $targetLocationId);
                    } elseif ($delta < 0) {
                        $inventoryService->deductStock($product->fresh(), abs($delta));
                    }

                    $receiptItem->update([
                        'status'         => 'reconciled',
                        'reconciled_qty' => $actualQty,
                        'nfe_key'        => $invoiceKey,
                    ]);

                    $receipt = $receiptItem->goodsReceipt;
                    if ($receipt) {
                        $hasPending = $receipt->items()->where('status', '!=', 'reconciled')->exists();
                        if (!$hasPending) {
                            $receipt->update([
                                'status'  => 'reconciled',
                                'nfe_key' => $invoiceKey,
                            ]);
                        }
                    }
                }
            } else {
                $targetLocationId = $this->resolveStorageLocationForProcurement($product);
                if ($product->track_inventory) { $inventoryService->addStock($product->fresh(), $actualQty, 'nfe_import', null, $targetLocationId); }
            }

            // 📡 Transmite a atualização do catálogo aos clientes conectados
            $this->dispatchCatalogUpdated();
        }

        ProductCostHistory::create([
            'product_id'               => $productId,
            'supplier_id'              => $supplierId,
            'nfe_key'                  => $invoiceKey,
            'xml_name'                 => $item['xml_name'],
            'quantity'                 => $actualQty,
            'unit_cost'                => $actualUnitCost,
            'conversion_factor'        => $conversionFactor,
            'original_quantity'        => $processQty,
            'original_unit_cost'       => $invoiceCost,
            'original_unit_of_measure' => $invoiceUom,
            'auto_approve'             => $autoApprove,
        ]);
    }

    public function processConfirmedItems(array $payload, array $currentItems): array
    {
        $processQty = (float) ($payload['xml_qty'] ?? 0);
        $availableQty = (float) ($payload['available_qty'] ?? 0);

        if ($processQty <= 0 || $processQty > $availableQty) {
            throw new InvalidArgumentException("A quantidade informada ({$processQty}) deve ser maior que zero e menor ou igual a {$availableQty}.");
        }

        $supplierId = isset($payload['supplier_id']) && !empty($payload['supplier_id'])
            ? (int) $payload['supplier_id']
            : null;

        if (!$supplierId) {
            throw new InvalidArgumentException('O fornecedor é obrigatório para processar itens da nota.');
        }

        $destination = $payload['destination_type'] ?? (!empty($payload['is_personal']) ? 'personal' : 'stock');
        $productId = isset($payload['product_id']) && !empty($payload['product_id'])
            ? (int) $payload['product_id']
            : null;

        if ($destination === 'stock' && !$productId) {
            throw new InvalidArgumentException('Selecione um produto cadastrado para entrada em estoque.');
        }

        $invoiceKey = $payload['invoice_key'] ?? null;
        $targetXmlName = (string) ($payload['xml_name'] ?? '');
        $conversionFactor = (float) ($payload['conversion_factor'] ?? 1.0);
        $uuidToFind = (string) ($payload['item_uuid'] ?? '');
        $receiptItemId = $payload['goods_receipt_item_id'] ?? null;

        $itemData = [
            'xml_qty'               => $processQty,
            'xml_cost'              => (float) ($payload['xml_cost'] ?? 0),
            'xml_uom'               => (string) ($payload['xml_uom'] ?? ''),
            'destination_type'      => $destination,
            'is_personal'           => $destination === 'personal',
            'xml_name'              => $targetXmlName,
            'product_id'            => $productId,
            'goods_receipt_item_id' => $receiptItemId,
            'conversion_factor'     => $conversionFactor,
            'asset_name'            => $payload['asset_name'] ?? null,
            'asset_location'        => $payload['asset_location'] ?? null,
            'asset_tag'             => $payload['asset_tag'] ?? null,
            'warranty_until'        => $payload['warranty_until'] ?? null,
        ];

        return DB::transaction(function () use ($itemData, $invoiceKey, $supplierId, $currentItems, $uuidToFind, $processQty, $availableQty, $targetXmlName, $destination, $productId, $conversionFactor, $receiptItemId) {

            $this->processDirectItem($itemData, $invoiceKey, $supplierId, false);

            $autoProcessedCount = 0;
            $items = $currentItems;

            foreach ($items as $key => $item) {
                if ((isset($item['item_uuid']) && (string)$item['item_uuid'] === $uuidToFind) || (string)$key === $uuidToFind) {
                    if ($processQty < $availableQty) {
                        $newAvailable = round($availableQty - $processQty, 4);
                        $items[$key]['available_qty'] = $newAvailable;
                        $items[$key]['xml_qty'] = $newAvailable;
                        $items[$key]['destination_type'] = $destination;
                        $items[$key]['is_personal'] = ($destination === 'personal');
                        $items[$key]['factor_source'] = 'manual';
                    } else {
                        unset($items[$key]);
                    }
                    continue;
                }

                // Efeito dominó para itens idênticos na mesma nota
                if (!$receiptItemId && isset($item['xml_name']) && (string)$item['xml_name'] === $targetXmlName) {
                    $duplicateItemData = array_merge($itemData, [
                        'xml_qty'  => (float) ($item['available_qty'] ?? $item['xml_qty'] ?? 0),
                        'xml_cost' => (float) ($item['xml_cost'] ?? 0),
                        'xml_uom'  => (string) ($item['xml_uom'] ?? ''),
                    ]);

                    $this->processDirectItem($duplicateItemData, $invoiceKey, $supplierId, false);
                    unset($items[$key]);
                    $autoProcessedCount++;
                }
            }

            return [
                'updated_items'        => $items,
                'auto_processed_count' => $autoProcessedCount,
            ];
        });
    }

    /**
     * 📡 Transmite a atualização aos terminais (PDV, Cardápio, Team App) após commit do banco.
     */
    protected function dispatchCatalogUpdated(): void
    {
        $tenantId = function_exists('tenant') && tenant('id')
            ? tenant('id')
            : (tenancy()->initialized ? tenancy()->tenant?->id : null);

        if (!$tenantId) {
            return;
        }

        // Evita múltiplos broadcasts redundantes dentro da mesma requisição/transação
        $lockKey = 'catalog_updated_dispatched_' . $tenantId;
        if (app()->bound($lockKey)) {
            return;
        }
        app()->instance($lockKey, true);

        DB::afterCommit(function () use ($tenantId) {
            try {
                broadcast(new CatalogUpdated($tenantId));
            } catch (\Throwable $e) {
                Log::warning("Falha ao transmitir CatalogUpdated pós-NFe: " . $e->getMessage());
            }
        });
    }

public function attemptAutoApproval(ReceivedInvoice $invoice): bool
    {
        $data = $this->extractDataFromXml($invoice->xml_content, $invoice->id);
        if (empty($data['xml_items']) || empty($data['supplier_id'])) return false;

        $supplier = Supplier::find($data['supplier_id']);
        if (!$supplier || !$supplier->is_trusted) return false;

        $familyIds = !empty($supplier->family_ids) ? $supplier->family_ids : [$supplier->id];
        $itemsToProcess = [];

        foreach ($data['xml_items'] as $item) {
            $xmlUom = strtoupper(preg_replace('/[^A-Za-z]/', '', $item['xml_uom']));

            $stockRule = ProductCostHistory::whereIn('supplier_id', $familyIds)
                ->where('xml_name', $item['xml_name'])
                ->where('auto_approve', true)
                ->latest('id')
                ->first();

            if ($stockRule && $this->areUnitsCompatible($xmlUom, $stockRule->original_unit_of_measure)) {
                $itemsToProcess[] = [
                    'product_id'        => $stockRule->product_id,
                    'conversion_factor' => (float) $stockRule->conversion_factor,
                    'destination_type'  => 'stock',
                    'is_personal'       => false,
                    'item'              => $item,
                ];
                continue;
            }

            $personalRule = PersonalPurchase::whereIn('supplier_id', $familyIds)
                ->where('xml_name', $item['xml_name'])
                ->where('auto_approve', true)
                ->latest('id')
                ->first();

            if ($personalRule) {
                $itemsToProcess[] = [
                    'product_id'        => null,
                    'conversion_factor' => 1.0,
                    'destination_type'  => 'personal',
                    'is_personal'       => true,
                    'item'              => $item,
                ];
                continue;
            }

            \Illuminate\Support\Facades\Log::info("⏸️ [Auto-Aprovação Pendente] NF-e {$invoice->nfe_key}: O item '{$item['xml_name']}' requer conferência.");
            return false;
        }

        \Illuminate\Support\Facades\DB::transaction(function () use ($itemsToProcess, $data, $invoice, $supplier) {
            $processor = app(InvoiceItemProcessor::class);
            foreach ($itemsToProcess as $p) {
                $payload = array_merge($p['item'], [
                    'product_id'        => $p['product_id'],
                    'conversion_factor' => $p['conversion_factor'],
                    'destination_type'  => $p['destination_type'],
                    'is_personal'       => $p['is_personal'],
                ]);
                $processor->processDirectItem($payload, $data['invoice_key'], $data['supplier_id'], true);
            }
            unset($data['xml_items']);
            $this->finalizeInvoice($data);
            \Illuminate\Support\Facades\Log::info("🚀 [Auto-Importação] NF-e {$invoice->nfe_key} finalizada automaticamente!");
        });

        return true;
    }

    private function areUnitsCompatible(?string $uomA, ?string $uomB): bool
    {
        $cleanA = strtoupper(preg_replace('/[^A-Za-z]/', '', (string) $uomA));
        $cleanB = strtoupper(preg_replace('/[^A-Za-z]/', '', (string) $uomB));
        if ($cleanA === $cleanB) return true;
        $synonyms = [
            'UN'  => ['UN', 'UND', 'UNID', 'PC', 'PCA'],
            'KG'  => ['KG', 'KIL', 'KILO', 'QUILO'],
            'LT'  => ['L', 'LT', 'LIT', 'LITRO'],
            'CX'  => ['CX', 'CXA', 'CAIXA'],
            'PCT' => ['PC', 'PCT', 'PACOTE'],
        ];
        foreach ($synonyms as $group) {
            if (in_array($cleanA, $group) && in_array($cleanB, $group)) return true;
        }
        return false;
    }
    public function isReadyToFinalize(ReceivedInvoice $invoice): bool
    {
        if ($invoice->status !== 'ready' || !$invoice->has_full_xml || empty($invoice->xml_content)) return false;
        $hasHistory = \App\Models\ProductCostHistory::where('nfe_key', $invoice->nfe_key)->exists()
            || \App\Models\PersonalPurchase::where('nfe_key', $invoice->nfe_key)->exists()
            || \App\Models\CompanyAsset::where('nfe_key', $invoice->nfe_key)->exists();
        if (!$hasHistory) return false;
        $data = $this->extractDataFromXml($invoice->xml_content, $invoice->id);
        return !empty($data) && empty($data['xml_items']) && !empty($data['supplier_id']);

    private static function getProductSelectField(): Select
    {
        return Select::make("product_id")
            ->label("Produto / Insumo / Suprimento")
            ->searchable()
            ->allowHtml()
            ->visible(fn(Get $get) => ($get("destination_type") ?? 'stock') === 'stock')
            ->required(fn(Get $get) => ($get("destination_type") ?? 'stock') === 'stock')
            ->getSearchResultsUsing(function (string $search): array {
                $products = Product::with('category')
                    ->whereIn('type', [\App\Enums\ProductType::INGREDIENT->value, \App\Enums\ProductType::RESALE->value, \App\Enums\ProductType::PACKAGING->value, \App\Enums\ProductType::CONSUMABLE->value])
                    ->where('name', 'like', "%{$search}%")
                    ->whereIn('status', ['available', 'out_of_stock', 'active'])
                    ->limit(50)
                    ->get();

                return $products->mapWithKeys(function ($product) {
                    return [$product->id => ProductFormatter::getHtmlLabel($product)];
                })->toArray();
            })
            ->getOptionLabelUsing(function ($value): ?string {
                $product = Product::with('category')->find($value);
                return $product ? ProductFormatter::getCleanLabel($product) : null;
            })
            ->live()
            ->afterStateUpdated(function ($state, Set $set, Get $get, $livewire) {
                if (!$state) return;

                $invoice = method_exists($livewire, 'getInvoiceRecord') ? $livewire->getInvoiceRecord() : null;
                $supplierId = $invoice?->supplier_id ?? ($livewire->data['supplier_id'] ?? null);

                $processor = app(InvoiceItemProcessor::class);
                $conversion = $processor->calculateSmartConversion(
                    productId: (int) $state,
                    supplierId: $supplierId ? (int) $supplierId : null,
                    currentXmlName: (string) $get('xml_name'),
                    xmlUomRaw: (string) $get('xml_uom')
                );

                $set('conversion_factor', $conversion['factor']);
                $set('factor_source', $conversion['source']);
            })
            ->hint(fn(Get $get) => $get("product_id") ? "U.M: " . Product::where("id", $get("product_id"))->value("unit_of_measure") : null)
            ->columnSpan(fn(Get $get) => empty($get('product_id')) ? 3 : 2)
            ->createOptionForm(
                \App\Filament\Resources\Products\Schemas\ProductForm::getQuickCreateSchema()
            )
            ->createOptionUsing(function (array $data, $livewire, Get $get) {
                $data["current_stock"] = 0;
                $data["base_cost"] = (float) $get("xml_cost") ?: 0;
                $product = Product::create($data);
                $livewire->dispatch('$refresh');
                return $product->id;
            })
            ->createOptionAction(fn(Action $action) => $action->modalHeading('Cadastro Rápido de Insumo')->modalWidth('sm'));
    }

    private static function getGoodsReceiptMatchField(): Select

            ->placeholder('Nenhum')
            ->options(function (Get $get, $livewire) {
                $productId = $get('product_id');

                $query = GoodsReceiptItem::with('goodsReceipt')

    }

    private static function getConfirmItemAction(): Actions
    {
        return Actions::make([
            Action::make("confirmItem")
                ->label(function (Get $get) {
                    return match ($get("destination_type")) {
                        'asset'    => 'Tombá-lo',
                        'personal' => 'Gerar Despesa',
                        default    => 'Associar',
                    };
                })
                ->icon(function (Get $get) {
                    return match ($get("destination_type")) {
                        'asset'    => 'heroicon-m-building-storefront',
                        'personal' => 'heroicon-m-banknotes',
                        default    => 'heroicon-m-link',
                    };
                })
                ->color(function (Get $get) {
                    return match ($get("destination_type")) {
                        'asset'    => 'warning',
                        'personal' => 'danger',
                        default    => 'primary',
                    };
                })
                ->size('sm')
                ->extraAttributes(['class' => 'whitespace-nowrap'])
                ->disabled(function (Get $get) {
                    $dest = $get("destination_type") ?? 'stock';
                    if ($dest === 'stock') {
                        return empty($get("product_id"));
                    }
                    if ($dest === 'asset') {
                        return empty($get("asset_name"));
                    }
                    return false;
                })
                ->action(function (Get $get, $livewire) {
                    $destination = $get('destination_type') ?? 'stock';
                    $processQty = (float) ($get('xml_qty') ?? 0);
                    $availableQty = (float) ($get('available_qty') ?? 0);

                    if ($processQty <= 0 || $processQty > $availableQty) {
                        Notification::make()
                            ->title('Quantidade Inválida')
                            ->body("A quantidade deve ser maior que zero e menor ou igual a {$availableQty}.")
                            ->danger()
                            ->send();
                        return;
                    }

                    $invoice = method_exists($livewire, 'getInvoiceRecord') ? $livewire->getInvoiceRecord() : null;
                    $supplierId = $invoice?->supplier_id ?? ($livewire->data['supplier_id'] ?? null);

                    if (!$supplierId) {
                        Notification::make()
                            ->title('Fornecedor Obrigatório')
                            ->body('Não foi possível identificar o fornecedor vinculado à nota.')
                            ->danger()
                            ->send();
                        return;
                    }

                    $productId = $get('product_id') ? (int) $get('product_id') : null;
                    if ($destination === 'personal') {
                        $productId = null;
                    }

                    $payload = [
                        'item_uuid'             => (string) ($get('item_uuid') ?? ''),
                        'xml_name'              => (string) ($get('xml_name') ?? ''),
                        'xml_qty'               => $processQty,
                        'available_qty'         => $availableQty,
                        'xml_cost'              => (float) ($get('xml_cost') ?? 0),
                        'xml_uom'               => (string) ($get('xml_uom') ?? ''),
                        'destination_type'      => $destination,
                        'is_personal'           => $destination === 'personal',
                        'conversion_factor'     => (float) ($get('conversion_factor') ?? 1),
                        'product_id'            => $productId,
                        'goods_receipt_item_id' => $get('goods_receipt_item_id') ? (int) $get('goods_receipt_item_id') : null,
                        'supplier_id'           => (int) $supplierId,
                        'invoice_key'           => $invoice?->nfe_key ?? $invoice?->access_key ?? ($livewire->data['invoice_key'] ?? null),
                        // Campos de Ativo Imobilizado
                        'asset_name'            => (string) ($get('asset_name') ?? $get('xml_name')),
                        'asset_location'        => $get('asset_location'),
                        'asset_tag'             => $get('asset_tag'),
                        'warranty_until'        => $get('warranty_until'),
                    ];

                    try {
                        $result = $livewire->confirmItem($payload);
                        $autoCount = $result['auto_processed_count'] ?? 0;
                        $targetXmlName = $payload['xml_name'] ?: 'Item';

                        $body = match ($destination) {
                            'asset'    => "Ativo Imobilizado '{$payload['asset_name']}' registrado no Patrimônio sem gerar estoque físico.",
                            'personal' => "Despesa pessoal gerada para '{$targetXmlName}'.",
                            default    => "Item '{$targetXmlName}' integrado ao estoque com sucesso!",
                        };

                        if ($destination === 'stock' && $payload['goods_receipt_item_id']) {
                            $body .= " (Estoque conciliado com o romaneio anterior).";
                        }

                        if ($autoCount > 0) {
                            $body .= " Efeito dominó: +{$autoCount} item(ns) idêntico(s) associado(s).";
                        }

                        $title = match ($destination) {
                            'asset'    => 'Patrimônio Registrado',
                            'personal' => 'Despesa Registrada',
                            default    => 'Estoque Conciliado',
                        };

                        Notification::make()
                            ->title($title)
                            ->body($body)
                            ->success()
                            ->send();

                    } catch (Throwable $e) {
                        Notification::make()
                            ->title('Erro ao Processar Item')
                            ->body($e->getMessage())
                            ->danger()
                            ->send();
                    }
                }),
        ])
        ->alignEnd()
        ->columnSpan(1)
        ->extraAttributes(["style" => "margin-top: 32px"]);
    }

    private static function getAssistantAction(): Action
    {
        return Action::make("assistente")
            ->icon("heroicon-m-sparkles")
            ->tooltip("Assistente de Conversão")
            ->color("primary")
            ->visible(fn(Get $get) => filled($get("product_id")) && ($get("destination_type") ?? 'stock') === 'stock')
            ->modalHeading("Assistente de Conversão")
            ->form(function (Get $get) {
                $xmlUom = strtoupper(preg_replace("/[^A-Z]/", "", (string) ($get("xml_uom") ?? "")));
                $internalUom = Product::find($get("product_id"))->unit_of_measure ?? "UN";
                $isPackage = in_array($xmlUom, ["CX", "FD", "FDO", "PC", "PCT", "SC"]);

                $fields = [Placeholder::make("info")->hiddenLabel()->content(new HtmlString("Nota: <strong>{$xmlUom}</strong> | Estoque: <strong>{$internalUom}</strong>"))];
                if ($isPackage) {
                    $fields[] = TextInput::make("qtd_pacote")->label("Itens por pacote/caixa")->numeric()->required();
                }
                if (in_array($xmlUom, ["KG", "KIL", "L", "LT", "LIT"]) && $internalUom === "UN") {
                    $fields[] = TextInput::make("peso_in")->label("Peso/Volume de 1 {$internalUom} (em {$xmlUom})")->numeric()->required();
                }
                if (($xmlUom === "UN" || $isPackage) && in_array($internalUom, ["KG", "G", "L", "ML"])) {
                    $fields[] = TextInput::make("peso_out")->label("Peso/Volume de 1 UN (em {$internalUom})")->numeric()->required();
                }

                return $fields;
            })
            ->action(function (array $data, Set $set, Get $get) {
                $xmlUom = strtoupper(preg_replace("/[^A-Z]/", "", (string) ($get("xml_uom") ?? "")));
                $internalUom = Product::find($get("product_id"))->unit_of_measure ?? "UN";
                $isPackage = in_array($xmlUom, ["CX", "FD", "FDO", "PC", "PCT", "SC"]);

                $qtd = (float) ($data["qtd_pacote"] ?? 1);
                $pesoIn = (float) ($data["peso_in"] ?? 0);
                $pesoOut = (float) ($data["peso_out"] ?? 0);

                $factor = 1.0;
                if ($xmlUom === $internalUom || (in_array($xmlUom, ["KG", "KIL"]) && $internalUom === "KG") || (in_array($xmlUom, ["L", "LT", "LIT"]) && $internalUom === "LT")) {
                    $factor = 1.0;
                } elseif (in_array($xmlUom, ["KG", "KIL"]) && $internalUom === "G") {
                    $factor = 1000.0;
                } elseif ($xmlUom === "G" && in_array($internalUom, ["KG", "KIL"])) {
                    $factor = 0.001;
                } elseif (in_array($xmlUom, ["L", "LT", "LIT"]) && $internalUom === "ML") {
                    $factor = 1000.0;
                } elseif ($xmlUom === "ML" && in_array($internalUom, ["L", "LT", "LIT"])) {
                    $factor = 0.001;
                } elseif ($isPackage && $internalUom === "UN") {
                    $factor = $qtd;
                } elseif (in_array($xmlUom, ["KG", "KIL", "L", "LT", "LIT"]) && $internalUom === "UN" && $pesoIn > 0) {
                    $factor = 1 / $pesoIn;
                } elseif ($isPackage && in_array($internalUom, ["KG", "G", "L", "ML"]) && $pesoOut > 0) {
                    $factor = $qtd * $pesoOut;
                } elseif ($xmlUom === "UN" && in_array($internalUom, ["KG", "G", "L", "ML"]) && $pesoOut > 0) {
                    $factor = $pesoOut;
                }

                $set("conversion_factor", round($factor, 4));
                $set("factor_source", "assistant_used");

                Notification::make()->title("Fator de Conversão Aplicado!")->success()->send();
            });
    }
